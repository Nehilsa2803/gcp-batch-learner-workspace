with Ref as (
select category, sub_category,sum(sales_amount) as sub_cat_sales
from ex_products p 
join ex_transactions t
on p.product_id = t.product_id
join ex_orders o 
on o.order_id = t.Order_ID
where o.order_status not in ('canceled', 'unavailable')
group by category, sub_category
)

select category,
sub_category,
sub_cat_sales,
dense_rank() over (partition by category order by sub_cat_sales desc)  as sub_cat_rank,
round(100.0 * sub_cat_sales / sum(sub_cat_sales) over (partition by category), 2) as sales_contribution_to_category,
round(100.0 * sub_cat_sales / sum(sub_cat_sales  ) over(), 2) as overall_sales_contribution
from Ref
;
