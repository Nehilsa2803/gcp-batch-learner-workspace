— q1

with cte as (
    select p.category, p.sub_category, round(sum(t.sales_amount), 2) as total_sales
    from ex_products p
    join ex_transactions t
    on p.product_id = t.product_id
    group by p.category, p.sub_category
),
sales_report as (
    select category, sub_category,
           rank() over (partition by category order by total_sales desc) as "sub_cat Rank",
           concat(round(total_sales * 100.0 / sum(total_sales) over (partition by category), 2), '%') as sales_contribution_to_category,
           concat(round(total_sales * 100.0 / sum(total_sales) over (), 2), '%') as overall_sales_contribution
    from cte
)       
select category, sub_category, sales_contribution_to_category
from sales_report
where sub_category = 'Appliances'
order by category, "sub_cat Rank";
