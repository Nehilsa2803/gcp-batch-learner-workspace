Question 1 : select name,
category,
price,
stock_quantity 
from products 
where stock_quantity < 10 
order by stock_quantity;

Question 2 : select 
o.payment_method,
sum(p.price*oi.quantity*(100-oi.discount_pct)/100.0) as total_revenue 
from orders o 
join order_items oi 
on o.order_id=oi.order_id 
join products p 
on p.product_id=oi.product_id 
where o.status='Delivered' 
group by o.payment_method ;

Question 3 : select c.full_name,
c.city,
c.account_type,
count(distinct o.order_id) 
from customers c join orders o 
on c.customer_id=o.customer_id 
group by 
c.full_name,
c.city,
c.account_type  
having count(distinct o.order_id)>1 ;


Question 4 : WITH ProductRevenue AS (
SELECT 
p.name,
p.category,
SUM(p.price * oi.quantity * (100 - oi.discount_pct) / 100.0) AS revenue 
FROM products p 
JOIN 
order_items oi 
ON p.product_id = oi.product_id 
JOIN 
orders o 
ON o.order_id = oi.order_id 
WHERE o.status = 'Delivered' 
GROUP BY p.product_id, 
p.name, p.category)

SELECT name,
category,
revenue,
dense_rank() OVER (PARTITION BY category ORDER BY revenue DESC) AS category_rank,
ROUND((revenue * 100.0 / SUM(revenue) OVER (PARTITION BY category)), 2) AS pct_of_category_total 
FROM ProductRevenue 
ORDER BY category,
category_rank;
