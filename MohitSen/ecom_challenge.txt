Q1. Retrieve name, category, price, and stock quantity for low-stock products, sorted by stock ascending.

select name, category, price, stock_quantity
from products
where stock_quantity < 10
order by stock_quantity 

Q2. For each payment method, calculate total revenue from delivered orders only.

select o.payment_method, round(sum((p.price * oi.quantity) * (1 - oi.discount_pct/100.0)), 2)
from products p 
join order_items oi on p.product_id = oi.product_id
join orders o on o.order_id = oi.order_id
where o.status = 'Delivered'
group by payment_method

Q3. Show qualifying customer name, city, account type, and total orders placed.

select c.full_name, c.city, c.account_type, count(o.order_id) as total_ord
from customers c join orders o
on c.customer_id = o.customer_id
group by c.full_name, c.city, c.account_type
Having total_ord > 1

Q4. For each product, show name, category, revenue, category rank, and percent of category total rounded to 2 decimals.

WITH product_revenue AS (
    SELECT
        p.product_id,
        p.name,
        p.category,
        SUM((p.price * oi.quantity) * (1 - oi.discount_pct / 100.0)) AS revenue
    FROM products p
    JOIN order_items oi
        ON p.product_id = oi.product_id
    GROUP BY
        p.product_id,
        p.name,
        p.category
),

category_revenue AS (
    SELECT
        category,
        SUM(revenue) AS total_category_revenue
    FROM product_revenue
    GROUP BY category
)

SELECT
    pr.name,
    pr.category,
    ROUND(pr.revenue, 2) AS revenue,
    RANK() OVER (
        PARTITION BY pr.category
        ORDER BY pr.revenue DESC
    ) AS category_rank,
    ROUND(
        pr.revenue * 100.0 / cr.total_category_revenue,
        2
    ) AS category_pct
FROM product_revenue pr
JOIN category_revenue cr
    ON pr.category = cr.category
ORDER BY
    pr.category,
    category_rank;
